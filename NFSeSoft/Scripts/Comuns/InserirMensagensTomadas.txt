function GetMensagemTomadas(const aValue: string): string;
  var
    _ListaMensagens : TStringList;
    _Mensagem : string;
  begin
    try
      _ListaMensagens := TStringList.Create;
      _ListaMensagens.LoadFromFile(Params.Values['DiretorioEsquemas'] + 'TecnoNFSe\ListaMensagensTomadas.txt');

      _Mensagem := _ListaMensagens.Values[Params.Values['Cidade']];
	 
      if Trim(_Mensagem) = '' then
        _Mensagem := _ListaMensagens.Values[Params.Values['Padrao']]; 
	
      if (Trim(aValue) <> '') then
		Result := aValue
      else if (Trim(_Mensagem) <> '') then 
        Result := _Mensagem
	  else 
		Result := 'Consulta não retornou nota com o período informado.';
		
    finally
      _ListaMensagens.Free;
    end;
  end;
