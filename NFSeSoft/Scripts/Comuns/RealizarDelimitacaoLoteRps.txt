  function RealizarDelimitacaoLoteRps(const aXML, aTagRps, aTagNodeAcimaRps: string): string;
  var
    _DelimitadorRPS, _XmlRps, _XmlAcima, _TagRps, _TagFechaAcima: string;
  begin
    Result := aXML;
    _DelimitadorRPS := Params.Values['DelimitadorRPS'];

    if _DelimitadorRPS = '' then
      _DelimitadorRPS := Params.Values['Delimitador'];

    if _DelimitadorRPS <> '' then
    begin
      Progress('Realizando a delimitação do lote de RPS');
      _XmlRps := XmlSelect(aXML, '//*[local-name()="' + aTagRps + '"]');
      _XmlAcima := XmlSelect(aXML, '//*[local-name()="' + aTagNodeAcimaRps + '"]');
      if (_XmlRps <> '') and (_XmlAcima <> '') then
      begin
        _TagRps := '<' + aTagRps;
        Result := StringReplace(Result, _TagRps, _DelimitadorRps + _TagRps, [rfReplaceAll]);
        _TagFechaAcima := '</' + aTagNodeAcimaRps;
        Result := StringReplace(Result, _TagFechaAcima, _DelimitadorRps + _TagFechaAcima, []);
      end
      else
        RaiseException('Erro durante a delimitação do lote de RPS: não foi encontrado Rps no lote');
    end;
  end;
  
  function RealizarDelimitacaoListaRps(const aOutput: string): string;
  var
    _DelimitadorRPS : string;
  begin
    Result := aOutput;
    if Params.Values['DelimitadorRPS'] <> '' then
      _DelimitadorRPS := Params.Values['DelimitadorRPS']
    else
      _DelimitadorRPS := Params.Values['Delimitador'];

    if _DelimitadorRPS <> '' then
    begin
      Result := StringReplace(Result, '<ListaRps><Rps>', '<ListaRps>' + _DelimitadorRPS + '<Rps>', [rfReplaceAll]);
      Result := StringReplace(Result, '</Rps><Rps>', '</Rps>' + _DelimitadorRPS + '<Rps>', [rfReplaceAll]);
      Result := StringReplace(Result, '</Rps></ListaRps>',  '</Rps>' + _DelimitadorRPS +
        '</ListaRps>', [rfReplaceAll]);
    end;
  end;
