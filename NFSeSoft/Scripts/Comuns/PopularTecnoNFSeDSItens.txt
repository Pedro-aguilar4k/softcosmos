{$I ..\Comuns\ObterNomeValor.txt}

  procedure PopularTecnoNFSeDSItens;
  var
    _i: Integer;
  begin
    if (Pos('INCLUIRRPS', Uppercase(_Tx2.Text)) <= 0) or (Pos('SALVARRPS', Uppercase(_Tx2.Text)) <= 0) then
      RaiseException('Favor informar a seção INCLUIRRPS/SALVARRPS no TX2');

    for _i := _inicio to _Tx2.Count - 1 do
    begin
      _linha := Trim(_Tx2[_i]);
   
      // Adiciona numa lista temporária os campos personalizados por município que não fazem parte do TecnoNFSe
      if (_linha <> '') and (_linha[1] = '@') then
        _CamposMunicipio.Add(IntToStr(_DsTecnoAtual.RecordCount) + _linha);

      if (_linha <> '') and (_linha[1] <> ';') and (_linha[1] <> '@')then  // pular linhas em branco,comentários e campos personalizados
      begin
        if EhComando(_linha) then
        begin  
          case AnsiIndexText(_linha, ['INCLUIR', 'SALVAR', 'INCLUIRRPS', 'SALVARRPS', 'INCLUIRSERVICO', 'SALVARSERVICO', 'INCLUIRFORMAPAGAMENTO', 'SALVARFORMAPAGAMENTO', 'INCLUIRDEDUCAO', 'SALVARDEDUCAO', 'INCLUIRDESPESA', 'SALVARDESPESA', 'INCLUIRREFERENCIA', 'SALVARREFERENCIA', 'INCLUIRDOCUMENTOREFERENCIADO', 'SALVARDOCUMENTOREFERENCIADO']) of
            0:
            begin
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_LOTE);
              _DsTecnoAtual.Append;
            end;
            1:
            begin
              _DsTecnoAtual.Post;
              _DsTecnoAtual := nil;
            end;
            2:
            begin
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_NOTA);
              _DsTecnoAtual.Append;
            end;
            3:
            begin
              _DsTecnoAtual.Post;
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_LOTE);
            end;
            4:
            begin            
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_SERVICO);
              _DsTecnoAtual.Append;
            end;
            5:
            begin            
              _DsTecnoAtual.Post;
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_NOTA);
            end;
            6:
            begin            
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName('FormaPagamento');
              _DsTecnoAtual.Append;
            end;
            7:
            begin
              _DsTecnoAtual.Post;
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_NOTA);
            end;
            8:
            begin
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName('Deducao');
              _DsTecnoAtual.Append;
            end;
            9:
            begin
              _DsTecnoAtual.Post;
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_NOTA);
            end;
            10:
            begin
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName('despesa');
              _DsTecnoAtual.Append;
            end;
            11:
            begin              
              _DsTecnoAtual.Post;
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_NOTA);
            end;
            12:
            begin
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName('Referencia');
              _DsTecnoAtual.Append;
            end;
            13:
            begin              
              _DsTecnoAtual.Post;
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_NOTA);
            end;     
            14:
            begin
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName('DocumentoReferenciado');
              _DsTecnoAtual.Append;
            end;
            15:
            begin              
              _DsTecnoAtual.Post;
              _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_NOTA);
            end; 			
          else
            RaiseException('Comando desconhecido: ' + _linha);
          end;
        end
        else
        begin
          ObterNomeValor(_linha, _NomeCampo, _ValorCampo);
          if _DsTecnoAtual <> nil then
            Campo(_DsTecnoAtual, _NomeCampo).AsString := _ValorCampo
          else
            RaiseException('Favor atribuir valores a campos somente entre os comandos de inserção e gravação');
        end;
      end;
    end;
  end;
