{
  Variaveis externas disponiveis:
    PathXsd: string;
    MappingXsdFileName: string;
    Input: string;
    Params: TStringList;
    Datasets: TspdXSDDataset;
}
var
  _Tx2: TStringList;
  _CamposMunicipio: TStringList;
  _inicio: Integer;
  _PadraoTecnoNFSe: Boolean;
  _linha, _NomeCampo, _ValorCampo: string;
  _TipoIF, _InscricaoFederal, _SerieDps, _NumeroDps: string;
  _PastaTecnoNFSe: string;
  _DsTecnoAtual: TSpdXSDClientDataSet;
  _DsCidadeAtual: TSpdXSDClientDataSet;
  _Vlr: Double;
  _QtdeRps, _ValorInt: Integer;
  _VersaoManual: string;

const
  DSTECNO_LOTE = 'TecnoNFSeNacional';
  DSTECNO_NOTA = 'Dps';
  DSTECNO_DEDUCOES = 'Deducao';
  DSTECNO_NFSEREFERENCIA = 'NfseReferencia';
  DSTECNO_DOCUMENTOREFERENCIADO = 'DocumentoReferenciado';
  DSCIDADE_LOTE = 'DPS';
  DSCIDADE_NOTA = 'DPS';
  DSCIDADE_DEDUCOES = 'docDedRed';
  DSCIDADE_NFSEREFERENCIA = 'refNFSe';
  DSCIDADE_DOCUMENTOREFERENCIADO = 'documentos';

  procedure ExceptionValorInvalidoCampo(const aCampo: string);
  begin
    RaiseException('Valor inválido para o campo ' + aCampo);
  end;

begin
  Progress('Criando XSDDataset');
  CriarDatasets;

  _Tx2 := TStringList.Create;
  _CamposMunicipio := TStringList.Create;

  try
    Progress('Carregando conteudo/arquivo TX2');
    CarregarTX2;

    _DsTecnoAtual := nil;

    if _PadraoTecnoNFSe then
    begin
      Progress('Usando Padrao TecnoNFSe: preparando datasets');
      PrepararTecnoNFSeNacionalDatasets;
      // Passar dados para o Datasets, convertendo do padrao TecnoNFSe para o padrao da Cidade

      // Comecar com as notas
      Progress('Usando Padrao TecnoNFSe: carregando dados das notas');
      _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_NOTA);
      _DsCidadeAtual := Datasets.GetDataSetByName(DSCIDADE_NOTA);

      _QtdeRps := _DsTecnoAtual.RecordCount;

      _DsTecnoAtual.First;

      while not _DsTecnoAtual.Eof do
      begin
        _DsCidadeAtual.Append;

        // Detectar versao do XSD
        if Trim(Params.Values['VersaoManual']) = 'v1.01' then
          _VersaoManual := '1.01'
        else
          _VersaoManual := '1.00';

        SetarCampoValor('versao', _VersaoManual);

        _ValorCampo := Trim(CampoTecno('TipoEmitente'));
        case AnsiIndexText(_ValorCampo, ['1', '2', '3']) of
          0: _NomeCampo := 'CpfCnpjPrestador';
          1: _NomeCampo := 'CpfCnpjTomador';
          2: _NomeCampo := 'CpfCnpjIntermediario';
        else
          ExceptionValorInvalidoCampo('TipoEmitente. Valores aceitos: 1 - Prestador, 2 - Tomador, 3 - Intermediário.');
        end;

        SetarCampoValorTamanho('tpEmit', _ValorCampo, 1);

        _ValorCampo := RemoveMascara(CampoTecno(_NomeCampo));
        case Length(_ValorCampo) of
          11:
          begin
            _TipoIF := '1';
            _InscricaoFederal := PreencheZerosEsquerda(_ValorCampo, 14);
          end;
          14:
          begin
            _TipoIF := '2';
            _InscricaoFederal := _ValorCampo;
          end;
        else
          ExceptionValorInvalidoCampo(_NomeCampo + '. Valores aceitos: 11 - CPF, 14 - CNPJ');
        end;

        _NumeroDps := PreencheZerosEsquerda(Trim(CampoTecno('NumeroDps')), 15);
        _SerieDps := PreencheZerosEsquerda(Trim(CampoTecno('SerieDps')), 5);

        _ValorCampo := 'DPS' + RemoveMascara(CampoTecno('CodigoCidadeEmitente')) +
                       _TipoIF + _InscricaoFederal + _SerieDps + _NumeroDps;

        SetarCampoValorTamanhoObrigatorio('IdDPS', _ValorCampo, 45);

        SetarCampoValorTamanhoObrigatorio('tpAmb', Params.Values['Ambiente'], 1);

        if Trim(CampoTecno('DataEmissaoDps')) = '' then
          ExceptionValorInvalidoCampo('DataEmissaoDps. Deve ser preenchido no seguinte formato: "YYYY-MM-DD"T"hh:mm:ss"');

        _ValorCampo := FormatAnyDateTime(CampoTecno('DataEmissaoDps'), 'YYYY-MM-DD"T"hh:mm:ss') + '-03:00';
        SetarCampoValorTamanhoObrigatorio('dhEmi', _ValorCampo, 25);

        SetarCampoValorTamanho('verAplic', 'EmissorWeb_0.6.3.0', 20);

        VerificaSeEhNumero('serie', _SerieDps);
        SetarCampoValorTamanhoObrigatorio('serie', RemoverZerosEsquerda(_SerieDps), 5);

        VerificaSeEhNumero('nDPS', _NumeroDps);
        SetarCampoValorTamanhoObrigatorio('nDPS', RemoverZerosEsquerda(_NumeroDps), 15);

        if Trim(CampoTecno('Competencia')) = '' then
          ExceptionValorInvalidoCampo('Competencia. Deve ser preenchido no seguinte formato: "YYYY-MM-DD"');

        _ValorCampo := FormatAnyDateTime(CampoTecno('Competencia'), 'YYYY-MM-DD');
        SetarCampoValorTamanhoObrigatorio('dCompet', _ValorCampo, 10);

        SetarCampoValorTamanhoObrigatorio('cLocEmi', RemoveMascara(CampoTecno('CodigoCidadeEmitente')), 7);

        { Grupo de info da NFSe a ser substituida }
        SetarCampoValorTamanho('chSubstda', CampoTecno('ChaveNfseSubstituida'), 50);
        _ValorCampo := Trim(CampoTecno('CodMotivoNfseSubstituida'));
        case AnsiIndexText(_ValorCampo, ['', '1', '2', '3', '4', '5', '99']) of
          0: ; //IGNORA
          1, 2, 3, 4, 5: SetarCampoValorTamanho('cMotivo', CampoTecno('CodMotivoNfseSubstituida'), 1);
          6:
          begin
            SetarCampoValorTamanho('cMotivo', CampoTecno('CodMotivoNfseSubstituida'), 2);
            SetarCampoTamanhoMinimo('MotivoNfseSubstituida', 'xMotivo', 255, 15);
          end;
        else
          ExceptionValorInvalidoCampo('CodMotivoNfseSubstituida.  Valores aceitos: ' + #13#10 +
            '1 - Desenquadramento de NFS-e do Simples Nacional;' + #13#10 +
            '2 - Enquadramento de NFS-e no Simples Nacional;' + #13#10 +
            '3 - Inclusão Retroativa de Imunidade/Isenção para NFS-e;' + #13#10 +
            '4 - Exclusão Retroativa de Imunidade/Isenção para NFS-e;' + #13#10 +
            '5 - Rejeição de NFS-e pelo tomador ou pelo intermediário se responsável pelo recolhimento do tributo;' + #13#10 +
            '99 - Outros;');
        end;

        // Dados Prestador
        if CampoCidade('tpEmit') = '1' then
          _ValorCampo := _InscricaoFederal
        else
          _ValorCampo := Trim(RemoveMascara(CampoTecno('CpfCnpjPrestador')));

        case Length(_ValorCampo) of
          11: SetarCampoValor('CPFPrestador', _ValorCampo);
          14: SetarCampoValor('CNPJPrestador', _ValorCampo);
        else
        begin
          if Trim(CampoTecno('NifPrestador')) <> '' then
            SetarCampoValorTamanho('NIFPrestador', CampoTecno('NifPrestador'), 40)
          else if Trim(CampoTecno('CodigoNifPrestador')) <> '' then
            SetarCampoValorTamanho('cNaoNIFPrestador', CampoTecno('CodigoNifPrestador'), 1)
          else
            ExceptionValorInvalidoCampo('CpfCnpjPrestador. Valores aceitos: 11 - CPF, 14 - CNPJ');
        end;
        end;

        SetarCampoValorTamanho('CAEPFPrestador', CampoTecno('NumeroCAEPFPrestador'), 14);
        SetarCampoValorTamanho('IMPrestador', CampoTecno('InscricaoMunicipalPrestador'), 15);
        SetarCampoValorTamanho('xNomePrestador', CampoTecno('RazaoSocialPrestador'), 150);

        _ValorCampo := Trim(RemoveMascara(CampoTecno('CodigoCidadePrestador')));
        if _ValorCampo = '9999999' then
        begin
          SetarCampoValorTamanho('cPaisPrestador', CampoTecno('PaisPrestador'), 2);
          SetarCampoValorTamanho('cEndPostPrestador', CampoTecno('CepPrestador'), 11);
          SetarCampoValorTamanho('xCidadePrestador', CampoTecno('DescricaoCidadePrestador'), 60);
          SetarCampoValorTamanho('xEstProvRegPrestador', CampoTecno('RegiaoPrestador'), 60);
        end
        else
        begin
          SetarCampoValorTamanho('cMunPrestador', _ValorCampo, 7);
          SetarCampoValorTamanho('CEPPrestador', RemoveMascara(CampoTecno('CepPrestador')), 8);
          SetarCampoValorTamanho('xLgrPrestador', CampoTecno('EnderecoPrestador'), 255);
          SetarCampoValorTamanho('nroPrestador', CampoTecno('NumeroPrestador'), 60);
          SetarCampoValorTamanho('xCplPrestador', CampoTecno('ComplementoPrestador'), 156);
          SetarCampoValorTamanho('xBairroPrestador', CampoTecno('BairroPrestador'), 60);
        end;

        SetarCampoValorTamanho('fonePrestador', CampoTecno('TelefonePrestador'), 20);
        SetarCampoValorTamanho('emailPrestador', CampoTecno('EmailPrestador'), 80);

        _ValorCampo := Trim(CampoTecno('OptanteSimplesNacional'));
        case AnsiIndexText(_ValorCampo, ['', '1', '2', '3']) of
          0, 1: SetarCampoValor('opSimpNac', '1');
          2, 3: SetarCampoValor('opSimpNac', _ValorCampo);
        else
          ExceptionValorInvalidoCampo('OptanteSimplesNacional. Valores aceitos: ' + #13#10 +
            '1 - Não Optante;' + #13#10 +
            '2 - Optante - Microempreendedor Individual (MEI);' + #13#10 +
            '3 - Optante - Microempresa ou Empresa de Pequeno Porte (ME/EPP);');
        end;

        _ValorCampo := Trim(CampoTecno('RegimeApuracaoTributaria'));
        case AnsiIndexText(_ValorCampo, ['', '1', '2', '3']) of
          0: ; //IGNORA
          1, 2, 3: SetarCampoValor('regApTribSN', _ValorCampo);
        else
          ExceptionValorInvalidoCampo('RegimeEspecialTributacao. Valores aceitos: ' + #13#10 +
            '1 – Regime de apuração dos tributos federais e municipal pelo SN;' + #13#10 +
            '2 – Regime de apuração dos tributos federais pelo SN e o ISSQN pela NFS-e conforme respectiva legislação municipal do tributo;' + #13#10 +
            '3 – Regime de apuração dos tributos federais e municipal pela NFS-e conforme respectivas legilações federal e municipal de cada tributo;');
        end;

        _ValorCampo := Trim(CampoTecno('RegimeEspecialTributacao'));
        case AnsiIndexText(_ValorCampo, ['0', '1', '2', '3', '4', '5', '6', '']) of
          0, 7: SetarCampoValor('regEspTrib', '0');
          1, 2, 3, 4, 5, 6: SetarCampoValor('regEspTrib', _ValorCampo);
        else
          ExceptionValorInvalidoCampo('RegimeEspecialTributacao. Valores aceitos: ' + #13#10 +
            '0 - Nenhum;' + #13#10 +
            '1 - Ato Cooperado (Cooperativa);' + #13#10 +
            '2 - Estimativa;' + #13#10 +
            '3 - Microempresa Municipal;' + #13#10 +
            '4 - Notário ou Registrador;' + #13#10 +
            '5 - Profissional Autônomo;' + #13#10 +
            '6 - Sociedade de Profissionais;');
        end;

        // Dados Tomador
        if CampoCidade('tpEmit') = '2' then
          _ValorCampo := _InscricaoFederal
        else
          _ValorCampo := Trim(RemoveMascara(CampoTecno('CpfCnpjTomador')));

        case Length(_ValorCampo) of
          0: ; //IGNORA
          11: SetarCampoValor('CPFTomador', _ValorCampo);
          14: SetarCampoValor('CNPJTomador', _ValorCampo);
        else
          ExceptionValorInvalidoCampo('CpfCnpjTomador. Valores aceitos: 11 - CPF, 14 - CNPJ');
        end;

        if Trim(CampoTecno('NifTomador')) <> '' then
        begin
          SetarCampoValor('CPFTomador', '');
          SetarCampoValor('CNPJTomador', '');
          SetarCampoValorTamanho('NIFTomador', CampoTecno('NifTomador'), 40);
        end
        else if Trim(CampoTecno('CodigoNifTomador')) <> '' then
        begin
          SetarCampoValor('CPFTomador', '');
          SetarCampoValor('CNPJTomador', '');
          SetarCampoValorTamanho('cNaoNIFTomador', CampoTecno('CodigoNifTomador'), 1);
        end;

        SetarCampoValorTamanho('CAEPFTomador', CampoTecno('NumeroCAEPFTomador'), 14);
        SetarCampoValorTamanho('IMTomador', CampoTecno('InscricaoMunicipalTomador'), 15);
        SetarCampoValorTamanho('xNomeTomador', CampoTecno('RazaoSocialTomador'), 150);

        _ValorCampo := Trim(RemoveMascara(CampoTecno('CodigoCidadeTomador')));
        if _ValorCampo = '9999999' then
        begin
          SetarCampoValorTamanho('cPaisTomador', CampoTecno('PaisTomador'), 2);
          SetarCampoValorTamanho('cEndPostTomador', CampoTecno('CepTomador'), 11);
          SetarCampoValorTamanho('xCidadeTomador', CampoTecno('DescricaoCidadeTomador'), 60);
          SetarCampoValorTamanho('xEstProvRegTomador', CampoTecno('RegiaoTomador'), 60);
          SetarCampoValorTamanho('xBairroTomador', CampoTecno('BairroTomador'), 60);
          SetarCampoValorTamanho('nroTomador', CampoTecno('NumeroTomador'), 60);
          SetarCampoValorTamanho('xLgrTomador', CampoTecno('EnderecoTomador'), 255);
        end
        else
        begin
          SetarCampoValorTamanho('cMunTomador', _ValorCampo, 7);
          SetarCampoValorTamanho('CEPTomador', RemoveMascara(CampoTecno('CepTomador')), 8);
          SetarCampoValorTamanho('xLgrTomador', CampoTecno('EnderecoTomador'), 255);
          SetarCampoValorTamanho('nroTomador', CampoTecno('NumeroTomador'), 60);
          SetarCampoValorTamanho('xCplTomador', CampoTecno('ComplementoTomador'), 156);
          SetarCampoValorTamanho('xBairroTomador', CampoTecno('BairroTomador'), 60);
        end;

        SetarCampoValorTamanho('foneTomador', CampoTecno('TelefoneTomador'), 20);
        SetarCampoValorTamanho('emailTomador', CampoTecno('EmailTomador'), 80);

        // Dados Intermediario
        if CampoCidade('tpEmit') = '2' then
          _ValorCampo := _InscricaoFederal
        else
          _ValorCampo := Trim(RemoveMascara(CampoTecno('CpfCnpjIntermediario')));

        case Length(_ValorCampo) of
          0: ; //IGNORA
          11: SetarCampoValor('CPFIntermediario', _ValorCampo);
          14: SetarCampoValor('CNPJIntermediario', _ValorCampo);
        else
        begin
          if Trim(CampoTecno('NifIntermediario')) <> '' then
            SetarCampoValorTamanho('NIFIntermediario', CampoTecno('NifIntermediario'), 40)
          else if Trim(CampoTecno('CodigoNifIntermediario')) <> '' then
            SetarCampoValorTamanho('cNaoNIFIntermediario', CampoTecno('CodigoNifIntermediario'), 1)
          else
            ExceptionValorInvalidoCampo('CpfCnpjIntermediario. Valores aceitos: 11 - CPF, 14 - CNPJ');
        end;
        end;

        SetarCampoValorTamanho('CAEPFIntermediario', CampoTecno('NumeroCAEPFIntermediario'), 14);
        SetarCampoValorTamanho('IMIntermediario', CampoTecno('InscricaoMunicipalIntermediario'), 15);
        SetarCampoValorTamanho('xNomeIntermediario', CampoTecno('RazaoSocialIntermediario'), 150);

        _ValorCampo := Trim(RemoveMascara(CampoTecno('CodigoCidadeIntermediario')));
        if _ValorCampo = '9999999' then
        begin
          SetarCampoValorTamanho('cPaisIntermediario', CampoTecno('PaisIntermediario'), 2);
          SetarCampoValorTamanho('cEndPostIntermediario', CampoTecno('CepIntermediario'), 11);
          SetarCampoValorTamanho('xCidadeIntermediario', CampoTecno('DescricaoCidadeIntermediario'), 60);
          SetarCampoValorTamanho('xEstProvRegIntermediario', CampoTecno('RegiaoIntermediario'), 60);
        end
        else
        begin
          SetarCampoValorTamanho('cMunIntermediario', _ValorCampo, 7);
          SetarCampoValorTamanho('CEPIntermediario', RemoveMascara(CampoTecno('CepIntermediario')), 8);
          SetarCampoValorTamanho('xLgrIntermediario', CampoTecno('EnderecoIntermediario'), 255);
          SetarCampoValorTamanho('nroIntermediario', CampoTecno('NumeroIntermediario'), 60);
          SetarCampoValorTamanho('xCplIntermediario', CampoTecno('ComplementoIntermediario'), 156);
          SetarCampoValorTamanho('xBairroIntermediario', CampoTecno('BairroIntermediario'), 60);
        end;

        SetarCampoValorTamanho('foneIntermediario', CampoTecno('TelefoneIntermediario'), 20);
        SetarCampoValorTamanho('emailIntermediario', CampoTecno('EmailIntermediario'), 80);

        // Dados Servico
        SetarCampoValorTamanhoObrigatorio('cLocPrestacao', RemoveMascara(CampoTecno('CodigoCidadePrestacao')), 7);
        SetarCampoValorTamanho('cPaisPrestacao', RemoveMascara(CampoTecno('CodigoPaisPrestacao')), 2);

        SetarCampoValorTamanhoObrigatorio('cTribNac', RemoveMascara(CampoTecno('CodigoItemListaServico')), 6);
        SetarCampoValorTamanho('cTribMun', RemoveMascara(CampoTecno('CodigoTributacaoMunicipio')), 3);
        SetarCampoValorTamanhoObrigatorio('xDescServ', CampoTecno('DiscriminacaoServico'), 1000);
        SetarCampoValorTamanho('cNBS', RemoveMascara(CampoTecno('CodigoNbs')), 9);
        SetarCampoValorTamanho('cIntContrib', RemoveMascara(CampoTecno('CodigoContribuinte')), 20);

        _ValorCampo := Trim(CampoTecno('CodigoModoPrestacao'));
        case AnsiIndexText(_ValorCampo, ['0', '1', '2', '3', '4', '']) of
          0, 1, 2, 3, 4: SetarCampoValor('mdPrestacao', _ValorCampo);
          5: ; //IGNORA
        else
          ExceptionValorInvalidoCampo('CodigoModoPrestacao. Valores aceitos: ' + #13#10 +
            '0 - Desconhecido (tipo não informado na nota de origem);' + #13#10 +
            '1 - Transfronteiriço;' + #13#10 +
            '2 - Consumo no Brasil;' + #13#10 +
            '3 - Movimento Temporário de Pessoas Físicas;' + #13#10 +
            '4 - Consumo no Exterior;');
        end;

        _ValorCampo := Trim(CampoTecno('CodigoVinculoPrestacao'));
        case AnsiIndexText(_ValorCampo, ['0', '1', '2', '3', '4', '5', '6', '']) of
          0, 1, 2, 3, 4, 5, 6: SetarCampoValor('vincPrest', _ValorCampo);
          7: ; //IGNORA
        else
          ExceptionValorInvalidoCampo('CodigoVinculoPrestacao. Valores aceitos: ' + #13#10 +
            '0 - Sem vínculo com o tomador/ Prestador' + #13#10 +
            '1 - Controlada;' + #13#10 +
            '2 - Controladora;' + #13#10 +
            '3 - Coligada;' + #13#10 +
            '4 - Matriz;' + #13#10 +
            '5 - Filial ou sucursal;' + #13#10 +
            '6 - Outro vínculo;');
        end;

        SetarCampoValorTamanho('tpMoeda', CampoTecno('TipoMoedaEstrangeira'), 3);
        SetarCampoValorCurrency('vServMoeda', CampoTecno('ValorServicosEstrangeiro'), False);

        SetarCampoValorTamanho('mecAFComexP', CampoTecno('TipoFomentoPrestador'), 2);
        SetarCampoValorTamanho('mecAFComexT', CampoTecno('TipoFomentoTomador'), 2);

        _ValorCampo := Trim(CampoTecno('TipoMovimentacaoBens'));
        case AnsiIndexText(_ValorCampo, ['0', '1', '2', '3', '']) of
          0, 1, 2, 3: SetarCampoValor('movTempBens', _ValorCampo);
          4: ; //IGNORA
        else
          ExceptionValorInvalidoCampo('TipoMovimentacaoBens. Valores aceitos: ' + #13#10 +
            '0 - Desconhecido (tipo não informado na nota de origem);' + #13#10 +
            '1 - Não;' + #13#10 +
            '2 - Vinculada - Declaração de Importação;' + #13#10 +
            '3 - Vinculada - Declaração de Exportação;');
        end;

        SetarCampoValorTamanho('nDI', CampoTecno('NumeroDeclaracaoImp'), 2);
        SetarCampoValorTamanho('nRE', CampoTecno('NumeroRegistroExp'), 2);

        _ValorCampo := Trim(CampoTecno('TipoMDIC'));
        case AnsiIndexText(_ValorCampo, ['0', '1', '']) of
          0, 1: SetarCampoValor('mdic', _ValorCampo);
          2: ; //IGNORA
        else
          ExceptionValorInvalidoCampo('TipoMDIC. Valores aceitos: ' + #13#10 +
            '0 - Não enviar para o MDIC;' + #13#10 +
            '1 - Enviar para o MDIC;');
        end;

        if CampoTecno('CodigoObra') <> '' then
          SetarCampoValorTamanho('cObra', CampoTecno('CodigoObra'), 30)
        else if CampoTecno('CodigoInscImobFisc') <> '' then 
          SetarCampoValorTamanho('inscImobFisc', CampoTecno('CodigoInscImobFisc'), 30)
        else
        begin
          if CampoTecno('CepObra') <> '' then 
            SetarCampoValorTamanho('CEPServicoObra', CampoTecno('CepObra'), 8)
          else 
          begin 
            SetarCampoValorTamanho('cEndPostServicoObra', CampoTecno('CodigoEnderecoObra'), 11);
            SetarCampoValorTamanho('xCidadeServicoObra', CampoTecno('NomeCidadeObra'), 60);
            SetarCampoValorTamanho('xEstProvRegServicoObra', CampoTecno('RegiaoObra'), 60);
          end;
          SetarCampoValorTamanho('xLgrServicoObra', CampoTecno('EnderecoObra'), 255);
          SetarCampoValorTamanho('nroServicoObra', CampoTecno('NumeroEndObra'), 60);
          SetarCampoValorTamanho('xCplServicoObra', CampoTecno('ComplementoObra'), 156);
          SetarCampoValorTamanho('xBairroServicoObra', CampoTecno('BairroObra'), 60);
        end;


        if Trim(CampoTecno('IdEvento')) <> '' then

              // Se tiver CIB, usar CIB; senao usar endereco
              _ValorCampo := Trim(CampoTecno('CodigoCIB'));
              if _ValorCampo <> '' then
              begin
                SetarCampoValorTamanho('cCIBImovel', _ValorCampo, 8);
              end
              else
              begin
                // Endereco do Imovel
                _ValorCampo := Trim(CampoTecno('CodigoCidadeImovel'));
                if (_ValorCampo <> '') and (_ValorCampo <> '9999999') then
                begin
                  // Endereco Nacional
                  SetarCampoValorTamanho('CEPImovel', RemoveMascara(CampoTecno('CepImovel')), 8);
                end
                else
                begin
                  // Endereco Exterior
                  SetarCampoValorTamanho('cEndPostImovel', CampoTecno('CodigoPostalImovel'), 11);
                  SetarCampoValorTamanho('xCidadeImovel', CampoTecno('DescricaoCidadeImovel'), 60);
                  SetarCampoValorTamanho('xEstProvRegImovel', CampoTecno('UfImovel'), 60);
                end;

                SetarCampoValorTamanho('xLgrImovel', CampoTecno('EnderecoImovel'), 255);
                SetarCampoValorTamanho('nroImovel', CampoTecno('NumeroImovel'), 60);
                SetarCampoValorTamanho('xCplImovel', CampoTecno('ComplementoImovel'), 156);
                SetarCampoValorTamanho('xBairroImovel', CampoTecno('BairroImovel'), 60);
              end;
            end;

            // Tributacao IBSCBS
            _ValorCampo := Trim(CampoTecno('SituacaoTributariaIbsCbs'));
            if _ValorCampo <> '' then
            begin
              SetarCampoValorTamanho('CSTIbsCbs', _ValorCampo, 3);
              SetarCampoValorTamanho('cClassTrib', CampoTecno('ClassificacaoTributariaIbsCbs'), 6);
              SetarCampoValorTamanho('cCredPres', CampoTecno('CodigoCreditoPresumidoIbsCbs'), 2);

              // Regime Regular
              SetarCampoValorTamanho('sitTribRegular', CampoTecno('SituacaoTributariaRegularIbsCbs'), 3);
              SetarCampoValorTamanho('cClassTribReg', CampoTecno('ClassifTributariaRegularIbsCbs'), 6);

              // Diferimento
              SetarCampoValorCurrency('pDifUF', CampoTecno('ValorPercDiferimentoEstadual'), False);
              SetarCampoValorCurrency('pDifMun', CampoTecno('ValorPercDiferimentoMunicipal'), False);
              SetarCampoValorCurrency('pDifCBS', CampoTecno('ValorPercDiferimentoCbs'), False);
            end;

            // NFSe Referenciada IBSCBS
            _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_NFSEREFERENCIA);
            _DsTecnoAtual.First;
            while not _DsTecnoAtual.Eof do
            begin
              _DsCidadeAtual := Datasets.GetDataSetByName(DSCIDADE_NFSEREFERENCIA);
              _DsCidadeAtual.Append;

              SetarCampoValorTamanho('refNFSe', CampoTecno('ChaveNFSe'), 50);

              _DsCidadeAtual.Post;
              _DsTecnoAtual.Next;
            end;

            _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_NOTA);
            _DsCidadeAtual := Datasets.GetDataSetByName(DSCIDADE_NOTA);

            // Documentos Referenciados para reembolso/repasse/ressarcimento IBSCBS
            _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_DOCUMENTOREFERENCIADO);
            _DsTecnoAtual.First;
            while not _DsTecnoAtual.Eof do
            begin
              _DsCidadeAtual := Datasets.GetDataSetByName(DSCIDADE_DOCUMENTOREFERENCIADO);
              _DsCidadeAtual.Append;

              // Verificar qual tipo de documento preencher (xs:choice - apenas 1 dos 3 grupos)
              _ValorCampo := Trim(CampoTecno('ChaveDFe'));
              if _ValorCampo <> '' then
              begin
                SetarCampoValorTamanho('tipoChaveDFe', CampoTecno('TipoChaveDFe'), 1);
                SetarCampoValorTamanho('xTipoChaveDFe', CampoTecno('DescricaoTipoChaveDFe'), 255);
                SetarCampoValorTamanho('chaveDFe', _ValorCampo, 50);
              end
              else if Trim(CampoTecno('NumeroDocumentoFiscal')) <> '' then
              begin
                SetarCampoValorTamanho('cMunDocFiscal', CampoTecno('MunicipioDocumentoFiscal'), 7);
                SetarCampoValorTamanho('nDocFiscal', CampoTecno('NumeroDocumentoFiscal'), 255);
                SetarCampoValorTamanho('xDocFiscal', CampoTecno('DescricaoDocumentoFiscal'), 255);
              end
              else if Trim(CampoTecno('NumeroDocumentoNaoFiscal')) <> '' then
              begin
                SetarCampoValorTamanho('nDocNaoFiscal', CampoTecno('NumeroDocumentoNaoFiscal'), 255);
                SetarCampoValorTamanho('xDocNaoFiscal', CampoTecno('DescricaoDocumentoNaoFiscal'), 255);
              end;

              // Fornecedor do documento (opcional)
              _ValorCampo := Trim(RemoveMascara(CampoTecno('CpfCnpjFornecedorDocumento')));
              case Length(_ValorCampo) of
                14: SetarCampoValor('CNPJFornecedorDocRef', _ValorCampo);
                11: SetarCampoValor('CPFFornecedorDocRef', _ValorCampo);
              else
              begin
                _ValorCampo := Trim(CampoTecno('NifFornecedorDocumento'));
                if _ValorCampo <> '' then
                  SetarCampoValorTamanho('NIFFornecedorDocRef', _ValorCampo, 40)
                else
                begin
                  _ValorCampo := Trim(CampoTecno('CodigoNifFornecedorDocumento'));
                  if _ValorCampo <> '' then
                    SetarCampoValorTamanho('cNaoNIFFornecedorDocRef', _ValorCampo, 1);
                end;
              end;
              end;

              SetarCampoValorTamanho('xNomeFornecedorDocRef', CampoTecno('RazaoSocialFornecedorDocumento'), 150);

              SetarCampoValorTamanho('dtEmiDocDedutivel', CampoTecno('DataEmissaoDocumento'), 10);
              SetarCampoValorTamanho('dtCompDoc', CampoTecno('CompetenciaDocumento'), 10);
              SetarCampoValorTamanho('tpReeRepRes', CampoTecno('TipoOperacaoDocumento'), 2);
              SetarCampoValorTamanho('xTpReeRepRes', CampoTecno('DescricaoOperacao'), 150);
              SetarCampoValorCurrency('vlrReeRepRes', CampoTecno('ValorOperacao'), False);

              _DsCidadeAtual.Post;
              _DsTecnoAtual.Next;
            end;

            _DsTecnoAtual := TecnoNFSeDS.GetDataSetByName(DSTECNO_NOTA);
            _DsCidadeAtual := Datasets.GetDataSetByName(DSCIDADE_NOTA);
          end;
        end;

        _DsCidadeAtual.Post;
        _DsTecnoAtual.Next;
      end;
      SetarCamposMunicipio(False);
    end
    else
      Datasets.LoadFromTx2Content(_Tx2.Text);
  finally
    _Tx2.Free;
    _CamposMunicipio.Free;
    Progress('Arquivo TX2 carregado com Sucesso');
  end;
end.
